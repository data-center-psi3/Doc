-- =========================================================
-- BANCO DE DADOS - PROJETO DATA CENTERS
-- PostgreSQL
-- =========================================================

-- =========================================================
-- 1. TABELA DATA_CENTER
-- =========================================================

CREATE TABLE data_center (
    id SERIAL PRIMARY KEY,

    nome VARCHAR(150) NOT NULL,
    empresa VARCHAR(150),

    cidade VARCHAR(100) NOT NULL,
    estado VARCHAR(2) NOT NULL,
    pais VARCHAR(50) DEFAULT 'Brasil',

    latitude DECIMAL(10,7),
    longitude DECIMAL(10,7),

    descricao TEXT,

    potencia_mw DECIMAL(10,2),
    area_m2 DECIMAL(12,2),

    tipo_instalacao VARCHAR(100),

    status_validacao VARCHAR(30),
    data_validacao DATE
);


-- =========================================================
-- 2. TABELA FONTE
-- =========================================================

CREATE TABLE fonte (
    id SERIAL PRIMARY KEY,

    instituicao VARCHAR(150) NOT NULL,
    titulo VARCHAR(200) NOT NULL,

    url TEXT,

    data_publicacao DATE,
    data_consulta DATE NOT NULL
);


-- =========================================================
-- 3. TABELA INDICADOR_AMBIENTAL
-- =========================================================

CREATE TABLE indicador_ambiental (
    id SERIAL PRIMARY KEY,

    data_center_id INTEGER NOT NULL,

    tipo VARCHAR(100) NOT NULL,

    valor DECIMAL(15,4),
    unidade VARCHAR(50),

    periodo VARCHAR(30),

    fonte_id INTEGER,

    observacoes TEXT,

    CONSTRAINT fk_indicador_data_center
        FOREIGN KEY (data_center_id)
        REFERENCES data_center(id),

    CONSTRAINT fk_indicador_fonte
        FOREIGN KEY (fonte_id)
        REFERENCES fonte(id)
);


-- =========================================================
-- 4. TABELA BENEFICIO
-- =========================================================

CREATE TABLE beneficio (
    id SERIAL PRIMARY KEY,

    data_center_id INTEGER NOT NULL,

    tipo VARCHAR(50) NOT NULL,

    descricao TEXT NOT NULL,

    fonte_id INTEGER,

    CONSTRAINT fk_beneficio_data_center
        FOREIGN KEY (data_center_id)
        REFERENCES data_center(id),

    CONSTRAINT fk_beneficio_fonte
        FOREIGN KEY (fonte_id)
        REFERENCES fonte(id)
);


-- =========================================================
-- 5. TABELA CONTEUDO_PESQUISA
-- =========================================================

CREATE TABLE conteudo_pesquisa (
    id SERIAL PRIMARY KEY,

    data_center_id INTEGER,

    titulo VARCHAR(200) NOT NULL,

    conteudo TEXT NOT NULL,

    categoria VARCHAR(50),

    fonte_id INTEGER NOT NULL,

    data_publicacao DATE,

    CONSTRAINT fk_conteudo_data_center
        FOREIGN KEY (data_center_id)
        REFERENCES data_center(id),

    CONSTRAINT fk_conteudo_fonte
        FOREIGN KEY (fonte_id)
        REFERENCES fonte(id)
);


-- =========================================================
-- 6. TABELA DE IMPORTAÇÃO
-- Guarda os dados brutos vindos do CSV
-- =========================================================

CREATE TABLE data_center_importacao (
    id_original TEXT,

    operador TEXT,
    unidade TEXT,

    municipio TEXT,
    uf TEXT,

    codigo_ibge TEXT,

    endereco TEXT,

    latitude TEXT,
    longitude TEXT,

    precisao_geo TEXT,

    tipo_instalacao TEXT,
    status_operacional TEXT,

    inicio_operacao TEXT,

    potencia_ti_mw_unidade TEXT,
    potencia_campus_mw TEXT,

    area_m2_unidade TEXT,

    redes_conectadas TEXT,
    ix_conectados TEXT,
    carriers TEXT,

    voltada_ia TEXT,

    resfriamento TEXT,
    fonte_energia TEXT,

    pue_declarado TEXT,
    wue_declarado TEXT,

    certificacoes TEXT,

    robustez_evidencia TEXT,

    fonte_url TEXT,

    data_levantamento TEXT
);


-- =========================================================
-- 7. RELACIONAMENTO ENTRE DATA CENTER E FONTE
-- N:N
-- =========================================================

CREATE TABLE data_center_fonte (
    data_center_id INTEGER NOT NULL,
    fonte_id INTEGER NOT NULL,

    PRIMARY KEY (data_center_id, fonte_id),

    CONSTRAINT fk_dcf_data_center
        FOREIGN KEY (data_center_id)
        REFERENCES data_center(id),

    CONSTRAINT fk_dcf_fonte
        FOREIGN KEY (fonte_id)
        REFERENCES fonte(id)
);


-- =========================================================
-- ÍNDICES ÚTEIS
-- Melhoram algumas consultas
-- =========================================================

CREATE INDEX idx_data_center_cidade
ON data_center(cidade);

CREATE INDEX idx_data_center_empresa
ON data_center(empresa);

CREATE INDEX idx_indicador_data_center
ON indicador_ambiental(data_center_id);

CREATE INDEX idx_indicador_tipo
ON indicador_ambiental(tipo);

CREATE INDEX idx_beneficio_data_center
ON beneficio(data_center_id);

CREATE INDEX idx_conteudo_data_center
ON conteudo_pesquisa(data_center_id);

CREATE INDEX idx_conteudo_categoria
ON conteudo_pesquisa(categoria);

CREATE INDEX idx_importacao_municipio
ON data_center_importacao(municipio);


-- =========================================================
-- CONSULTAS DE TESTE
-- =========================================================

-- Ver todos os Data Centers
SELECT *
FROM data_center;


-- Ver Data Centers por cidade
SELECT
    cidade,
    COUNT(*) AS quantidade
FROM data_center
GROUP BY cidade
ORDER BY cidade;


-- Ver indicadores ambientais
SELECT
    dc.nome AS data_center,
    ia.tipo,
    ia.valor,
    ia.unidade,
    ia.periodo
FROM indicador_ambiental ia
JOIN data_center dc
    ON ia.data_center_id = dc.id;


-- Ver benefícios
SELECT
    dc.nome AS data_center,
    b.tipo,
    b.descricao
FROM beneficio b
JOIN data_center dc
    ON b.data_center_id = dc.id;


-- Ver Data Centers e suas fontes
SELECT
    dc.nome,
    dc.cidade,
    f.instituicao,
    f.titulo,
    f.url
FROM data_center_fonte dcf
JOIN data_center dc
    ON dcf.data_center_id = dc.id
JOIN fonte f
    ON dcf.fonte_id = f.id
ORDER BY dc.cidade, dc.nome;


-- Ver conteúdos de pesquisa que podem ser usados pelo agente de IA
SELECT
    dc.nome AS data_center,
    cp.titulo,
    cp.categoria,
    cp.conteudo,
    f.url
FROM conteudo_pesquisa cp
LEFT JOIN data_center dc
    ON cp.data_center_id = dc.id
JOIN fonte f
    ON cp.fonte_id = f.id;


-- =========================================================
-- EXEMPLOS DE FILTRO REGIONAL
-- =========================================================

SELECT *
FROM data_center
WHERE cidade IN (
    'Hortolândia',
    'Campinas',
    'Sumaré',
    'Paulínia'
)
ORDER BY cidade, nome;


-- =========================================================
-- EXEMPLO DE CONSULTA PARA MAPA
-- =========================================================

SELECT
    id,
    nome,
    empresa,
    cidade,
    estado,
    latitude,
    longitude,
    potencia_mw,
    tipo_instalacao
FROM data_center
WHERE latitude IS NOT NULL
  AND longitude IS NOT NULL;


-- =========================================================
-- EXEMPLO DE CONSULTA PARA O DASHBOARD
-- =========================================================

SELECT
    cidade,
    COUNT(*) AS quantidade_data_centers,
    SUM(potencia_mw) AS potencia_total_mw
FROM data_center
GROUP BY cidade
ORDER BY quantidade_data_centers DESC;
